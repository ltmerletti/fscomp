15void report_format_bytes(int64_t bytes, char *output_buffer, size_t buffer_size) {
16 if (!output_buffer || buffer_size == 0) return;

18 if (bytes < 0) {
19 snprintf(output_buffer, buffer_size, "0 B");

21 }
22
23 const char *units[] = {"B", "KB", "MB", "GB", "TB", "PB"};

27 while (floating_bytes >= BYTES_PER_KIBIBYTE && unit_index < MAXIMUM_UNIT_INDEX) {

29 unit_index++;
30 }
31
32 if (unit_index == 0) snprintf(output_buffer, buffer_size, "%" PRId64 " B", bytes);

34}
