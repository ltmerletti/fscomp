58static bool is_abbreviated_jobs_option(const char *argument) {
59 if (!argument || strncmp(argument, "--", 2) != 0) return false;

61 size_t argument_name_length = strcspn(argument, "=");

71 bool is_add) {

78 size_t changed_count = 0;

90 } else {
91 printf("Not found in tracked list: %s\n", targets[i]);
92 }
93 }

96 char *save_path = config_path ? config_expand_path(config_path) : config_get_default_path();
97 if (config_save(config, save_path)) {
98 if (is_add) {
99 printf("Saved %zu tracked director%s to config.\n", config->target_count,
100 config->target_count == 1 ? "y" : "ies");
101 } else {
102 printf("Saved updated list (%zu director%s remaining).\n", config->target_count,
103 config->target_count == 1 ? "y" : "ies");
104 }
105 } else {
106 fprintf(stderr, "Warning: failed to save config file.\n");
107 }
