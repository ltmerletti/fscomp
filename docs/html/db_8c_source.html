56static void sqlite_extension_function(sqlite3_context *context, int argument_count, sqlite3_value **arguments) {
57 if (argument_count != 1 || sqlite3_value_type(arguments[0]) == SQLITE_NULL) {
58 sqlite3_result_text(context, "none", -1, SQLITE_STATIC);
59 return;
60 }
61 const unsigned char *path = sqlite3_value_text(arguments[0]);
62 if (!path) {
63 sqlite3_result_text(context, "none", -1, SQLITE_STATIC);
64 return;
65 }
66 const char *extension = filter_get_extension((const char *)path);
67 if (extension && extension[0] != '\0') sqlite3_result_text(context, extension, -1, SQLITE_TRANSIENT);
68 else sqlite3_result_text(context, "none", -1, SQLITE_STATIC);
69}
