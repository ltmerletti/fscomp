130static void truncate_display_path(const char *source_path, char *destination_buffer, size_t buffer_size,
131 size_t maximum_length) {
132 if (!source_path || !destination_buffer || buffer_size == 0) return;
133
134 size_t source_length = strlen(source_path);
135 if (source_length <= maximum_length && source_length < buffer_size) {
136 strncpy(destination_buffer, source_path, buffer_size - 1);
137 destination_buffer[buffer_size - 1] = 0;
138 return;
139 }
140
141 const char ellipsis[] = "...";
142 size_t ellipsis_length = sizeof(ellipsis) - 1;
143 if (maximum_length <= ellipsis_length || buffer_size <= ellipsis_length + 1) {
144 strncpy(destination_buffer, ellipsis, buffer_size - 1);
145 destination_buffer[buffer_size - 1] = 0;
146 return;
147 }
148
149 size_t tail_length = maximum_length - ellipsis_length;
150 if (tail_length >= buffer_size - ellipsis_length) tail_length = buffer_size - ellipsis_length - 1;
151
152 const char *tail_start = source_path + (source_length - tail_length);
153 const char *next_slash = strchr(tail_start, '/');
154 if (next_slash && (size_t)(next_slash - tail_start) < 12) tail_start = next_slash;
155
156 snprintf(destination_buffer, buffer_size, "%s%s", ellipsis, tail_start);
157}

162static void render_progress_update(scan_execution_context_t *context, const scan_statistics_t *statistics,
163 const char *current_path, bool force_render) {
164 if (!context || !statistics || context->quiet || context->silent) return;
165
166 pthread_mutex_lock(&g_progress_mutex);
167
168 int64_t now_milliseconds = current_time_milliseconds();
169 if (!force_render
170 && (now_milliseconds - context->last_progress_time_milliseconds < PROGRESS_UPDATE_INTERVAL_MILLISECONDS)) {
171 pthread_mutex_unlock(&g_progress_mutex);
172 return;
173 }
174 context->last_progress_time_milliseconds = now_milliseconds;
175
176 char saved_string[FORMATTED_BUFFER_CAPACITY];
177 int64_t files_scanned = read_statistic(&statistics->files_scanned);
178 int64_t files_compressed = read_statistic(&statistics->files_compressed);
179 report_format_bytes(read_statistic(&statistics->bytes_saved), saved_string, sizeof(saved_string));
180
181 if (context->is_interactive_terminal) {
182 char display_path[PROGRESS_LINE_BUFFER_CAPACITY] = {0};
183 if (current_path)
184 truncate_display_path(current_path, display_path, sizeof(display_path), PROGRESS_PATH_DISPLAY_LIMIT);
185 printf("\r\033[K[%s: %" PRId64 " scanned | %" PRId64 " compressed | %s saved] %s", context->operation_name,
186 files_scanned, files_compressed, saved_string, display_path);
187 fflush(stdout);
188 } else if (files_scanned > 0 && (files_scanned % PROGRESS_NON_TTY_LOG_INTERVAL == 0 || force_render)) {
189 printf("[%s] %" PRId64 " files scanned, %" PRId64 " compressed, %s saved...\n", context->operation_name,
190 files_scanned, files_compressed, saved_string);
191 fflush(stdout);
192 }
193 pthread_mutex_unlock(&g_progress_mutex);
194}

302 precompressed_record.compressor = "system";

315static bool prepare_file_for_compression(scan_worker_context_t *worker_context, file_work_item_t *file) {
316 const config_t *config = worker_context->config;
317 db_t *read_database = worker_context->read_database;
318 db_t *database = worker_context->write_database;
319 const scan_options_t *options = worker_context->options;
320 scan_statistics_t *statistics = worker_context->statistics;
321 scan_execution_context_t *context = worker_context->context;
322 const char *path = file->path;
323 const struct stat *file_stat = &file->file_stat;
324
325 if (g_scan_interrupted) return false;
326
327 increment_statistic(&statistics->files_scanned, 1);
328
329 filter_result_t filter_result = filter_evaluate(config, path, file_stat);
330 if (filter_result != FILTER_ACTION_PROCESS) {
331 increment_statistic(&statistics->files_skipped_filter, 1);
332 if (options->verbose) printf(" [SKIP] %s: %s\n", path, filter_result_string(filter_result));
333 else render_progress_update(context, statistics, path, false);
334 return false;
335 }
336
337 increment_statistic(&statistics->logical_bytes_processed, (int64_t)file_stat->st_size);
338
339 file_metadata_t existing_metadata;
340 db_lookup_result_t lookup_result =
341 read_database ? db_get_file_metadata(read_database, path, &existing_metadata) : DB_LOOKUP_NOT_FOUND;
342 if (lookup_result == DB_LOOKUP_ERROR) {
343 __atomic_store_n(&context->database_failed, true, __ATOMIC_RELAXED);
344 increment_statistic(&statistics->files_failed, 1);
345 return false;
346 }
347 bool found_in_database = lookup_result == DB_LOOKUP_FOUND;
348
349 if (found_in_database && !options->rehash_all) {
350 if (existing_metadata.size == (int64_t)file_stat->st_size
351 && existing_metadata.modification_time == (int64_t)file_stat->st_mtime) {
352 increment_statistic(&statistics->files_unchanged_stat, 1);
353 if (options->verbose) printf(" [UNCHANGED] %s: size and modification time match database\n", path);
354 else render_progress_update(context, statistics, path, false);
355 return false;
356 }
357 }
358
359 bool already_compressed = compressor_is_file_compressed(file_stat);
360 if (already_compressed && !options->rehash_all) {
361 process_precompressed_file(database, path, file_stat, options, statistics, context);
362 return false;
363 }
364
365 pthread_once(&g_file_mutex_once_control, initialize_file_mutexes);
366 pthread_mutex_t *file_lock = get_file_mutex((uint64_t)file_stat->st_dev, (uint64_t)file_stat->st_ino);
367 pthread_mutex_lock(file_lock);
368
369 struct stat refreshed_stat;
370 if (stat(path, &refreshed_stat) != 0) {
371 increment_statistic(&statistics->files_failed, 1);
372 pthread_mutex_unlock(file_lock);
373 return false;
374 }
375 file->file_stat = refreshed_stat;
376 file_stat = &file->file_stat;
377
378 if (compressor_is_file_compressed(file_stat) && !options->rehash_all) {
379 process_precompressed_file(database, path, file_stat, options, statistics, context);
380 pthread_mutex_unlock(file_lock);
381 return false;
382 }
383
384 char file_hash[SHA256_HEX_STRING_LENGTH] = {0};
385 if (!hasher_file_sha256(path, file_hash)) {
386 increment_statistic(&statistics->files_failed, 1);
387 if (options->verbose) fprintf(stderr, " [ERROR] %s: could not compute sha256\n", path);
388 render_progress_update(context, statistics, path, false);
389 pthread_mutex_unlock(file_lock);
390 return false;
391 }
392
393 if (found_in_database && strcmp(existing_metadata.sha256, file_hash) == 0) {
394 increment_statistic(&statistics->files_unchanged_hash, 1);
395 if (options->verbose) printf(" [VERIFIED] %s: hash matches %s\n", path, file_hash);
396 else render_progress_update(context, statistics, path, false);
397 if (!options->dry_run && database)
398 update_file_modification_time_and_batch_commit(database, path, file_stat->st_mtime, context);
399 pthread_mutex_unlock(file_lock);
400 return false;
401 }
402
403 if (options->dry_run) {
404 if (!options->quiet && !options->silent)
405 printf(" [DRY RUN] %s: eligible to compress, size %" PRId64 " bytes\n", path, (int64_t)file_stat->st_size);
406 increment_statistic(&statistics->files_compressed, 1);
407 render_progress_update(context, statistics, path, false);
408 pthread_mutex_unlock(file_lock);
409 return false;
410 }
411
412 memcpy(file->file_hash, file_hash, sizeof(file->file_hash));
413 file->file_lock = file_lock;
414 return true;
415}

420static void record_compression_result(scan_worker_context_t *worker_context, const file_work_item_t *file,
421 const compress_outcome_t *compression_outcome) {
422 const config_t *config = worker_context->config;
423 db_t *database = worker_context->write_database;
424 const scan_options_t *options = worker_context->options;
425 scan_statistics_t *statistics = worker_context->statistics;
426 scan_execution_context_t *context = worker_context->context;
427
428 file_record_t new_record;
429 memset(&new_record, 0, sizeof(new_record));
430 new_record.path = file->path;
431 new_record.size = file->file_stat.st_size;
432 new_record.modification_time = file->file_stat.st_mtime;
433 strncpy(new_record.sha256, file->file_hash, sizeof(new_record.sha256) - 1);
434 new_record.last_compressed_at = (int64_t)time(NULL);
435 new_record.compressor = config->compressor;
436
437 if (compression_outcome->result == COMPRESS_RESULT_COMPRESSED) {
438 increment_statistic(&statistics->files_compressed, 1);
439 increment_statistic(&statistics->bytes_saved, compression_outcome->bytes_saved);
440 new_record.physical_bytes = compression_outcome->physical_bytes_after;
441 new_record.compressed_bytes = compression_outcome->physical_bytes_after;
442 new_record.status = "compressed";
443 notify_file_compressed(context, file->path, compression_outcome->bytes_saved);
444 } else if (compression_outcome->result == COMPRESS_RESULT_INCOMPRESSIBLE) {
445 increment_statistic(&statistics->files_incompressible, 1);
446 new_record.physical_bytes = compression_outcome->physical_bytes_after;
447 new_record.compressed_bytes = compression_outcome->physical_bytes_after;
448 new_record.status = "incompressible";
449 if (options->verbose) printf(" [UNCHANGED] %s: afsctool reported no savings\n", file->path);
450 else render_progress_update(context, statistics, file->path, false);
451 } else {
452 increment_statistic(&statistics->files_failed, 1);
453 new_record.physical_bytes = compressor_get_physical_bytes(&file->file_stat);
454 new_record.compressed_bytes = new_record.physical_bytes;
455 new_record.status = "failed";
456 if (options->verbose) printf(" [FAILED] %s: %s\n", file->path, compression_outcome->error_message);
457 else render_progress_update(context, statistics, file->path, false);
458 }
459
460 if (database) record_file_and_batch_commit(database, &new_record, context);
461}

466static void compress_prepared_files(scan_worker_context_t *worker_context, file_work_item_t * const *prepared_files,
467 size_t prepared_file_count) {
468 if (prepared_file_count == 0) return;
469
470 const char *paths[FILE_TASK_BATCH_CAPACITY];
471 compress_outcome_t compression_outcomes[FILE_TASK_BATCH_CAPACITY];
472 for (size_t index = 0; index < prepared_file_count; index++)
473 paths[index] = prepared_files[index]->path;
474
475 int compression_worker_count = 0;
476 if (prepared_file_count >= MINIMUM_PARALLEL_COMPRESSION_FILE_COUNT) {
477 compression_worker_count = worker_context->options->worker_count;
478 if (compression_worker_count > MAXIMUM_FLAT_DIRECTORY_COMPRESSION_WORKER_COUNT)
479 compression_worker_count = MAXIMUM_FLAT_DIRECTORY_COMPRESSION_WORKER_COUNT;
480 }
481 compressor_compress(worker_context->config, compression_worker_count, paths, prepared_file_count,
482 compression_outcomes);
483
484 for (size_t index = 0; index < prepared_file_count; index++) {
485 record_compression_result(worker_context, prepared_files[index], &compression_outcomes[index]);
486 if (prepared_files[index]->file_lock) pthread_mutex_unlock(prepared_files[index]->file_lock);
487 }
488}

527static void process_file_batch(scan_worker_context_t *worker_context, file_batch_task_t *batch) {
528 file_work_item_t *prepared_files[FILE_TASK_BATCH_CAPACITY];
529 size_t prepared_file_count = 0;

533 if (prepare_file_for_compression(worker_context, file)) {

535 pthread_mutex_unlock(file->file_lock);

539 }
540 compress_prepared_files(worker_context, prepared_files, prepared_file_count);
541
542 for (size_t index = 0; index < batch->file_count; index++)
543 free(batch->files[index].path);
544 free(batch);

577static bool process_path_entries(scan_worker_context_t *worker_context, const char *path) {
578 char *path_arguments[] = {(char *)path, NULL};
579 FTS *file_tree = fts_open(path_arguments, FTS_COMFOLLOW | FTS_PHYSICAL | FTS_NOCHDIR, NULL);
580 if (!file_tree) {
581 if (worker_context->options->verbose) perror(path);
582 return false;
583 }
584
585 bool root_accessible = true;
586 bool processing_succeeded = true;
587 bool parallel = worker_context->work_queue != NULL;
588 bool batch_regular_files = false;
589 file_batch_task_t *file_batch = NULL;
590 FTSENT *entry = NULL;
591 while (!g_scan_interrupted && (entry = fts_read(file_tree)) != NULL) {
592 if (entry->fts_level == 0) {
593 if (entry->fts_info == FTS_F) {
594 process_single_file(worker_context, entry->fts_path, entry->fts_statp);
595 } else if (entry->fts_info == FTS_D && parallel) {
596 size_t regular_file_count = 0;
597 for (FTSENT *child = fts_children(file_tree, 0); child; child = child->fts_link) {
598 if (child->fts_info == FTS_F && ++regular_file_count >= FILE_TASK_BATCH_CAPACITY) {
599 batch_regular_files = true;
600 break;
601 }
602 }
603 } else if (entry->fts_info == FTS_DNR || entry->fts_info == FTS_ERR || entry->fts_info == FTS_NS) {
604 root_accessible = false;
605 if (worker_context->options->verbose)
606 fprintf(stderr, "Cannot access %s: %s\n", entry->fts_path, strerror(entry->fts_errno));
607 }
608 continue;
609 }
610
611 if (entry->fts_info == FTS_D) {
612 if (filter_should_skip_directory(worker_context->config, entry->fts_name, entry->fts_path)) {
613 fts_set(file_tree, entry, FTS_SKIP);
614 continue;
615 }
616 if (parallel) {
617 if (!dispatch_child_path(worker_context, entry->fts_path)) processing_succeeded = false;
618 fts_set(file_tree, entry, FTS_SKIP);
619 }
620 } else if (entry->fts_info == FTS_F) {
621 if (!batch_regular_files) {
622 process_single_file(worker_context, entry->fts_path, entry->fts_statp);
623 continue;
624 }
625
626 if (!file_batch) {
627 file_batch = calloc(1, sizeof(*file_batch));
628 if (!file_batch) {
629 processing_succeeded = false;
630 break;
631 }
632 file_batch->base.type = SCAN_TASK_FILE_BATCH;
633 }
634 file_work_item_t *candidate = &file_batch->files[file_batch->file_count];
635 candidate->path = strdup(entry->fts_path);
636 if (!candidate->path) {
637 processing_succeeded = false;
638 break;
639 }
640 candidate->file_stat = *entry->fts_statp;
641 file_batch->file_count++;
642 if (file_batch->file_count == FILE_TASK_BATCH_CAPACITY) {
643 dispatch_file_batch(worker_context, file_batch);
644 file_batch = NULL;
645 }
646 } else if (entry->fts_info == FTS_DNR || entry->fts_info == FTS_ERR || entry->fts_info == FTS_NS) {
647 if (entry->fts_level == 0) root_accessible = false;
648 if (worker_context->options->verbose)
649 fprintf(stderr, "Cannot access %s: %s\n", entry->fts_path, strerror(entry->fts_errno));
650 }
651 }
652
653 if (file_batch) process_file_batch(worker_context, file_batch);
654 fts_close(file_tree);
655 return root_accessible && processing_succeeded && !g_scan_interrupted;
656}

661static void *scan_worker(void *argument) {
662 scan_worker_context_t *worker_context = argument;
663 void *queued_task;
664 pthread_once(&g_file_mutex_once_control, initialize_file_mutexes);
665 while (queue_pop(worker_context->work_queue, &queued_task)) {
666 scan_task_t *task = queued_task;
667 if (task->type == SCAN_TASK_DIRECTORY) {
668 directory_task_t *directory_task = (directory_task_t *)task;
669 if (!g_scan_interrupted && !process_path_entries(worker_context, directory_task->path))
670 __atomic_store_n(worker_context->worker_failed, true, __ATOMIC_RELAXED);
671 free(directory_task);
672 } else {
673 process_file_batch(worker_context, (file_batch_task_t *)task);
674 }
675 complete_pending_task(worker_context);
676 }
677 return NULL;
678}

683bool scanner_execute(const config_t *config, db_t *database, const scan_options_t *options,
684 scan_statistics_t *output_statistics) {
685 if (!config || !options || !output_statistics) return false;
686 memset(output_statistics, 0, sizeof(*output_statistics));
687
688 int worker_count = options->worker_count > DEFAULT_WORKER_COUNT ? options->worker_count : DEFAULT_WORKER_COUNT;
689 if (worker_count > MAXIMUM_WORKER_COUNT) worker_count = MAXIMUM_WORKER_COUNT;
690
691 int64_t start_time_milliseconds = current_time_milliseconds();
692 int64_t start_time_unix = (int64_t)time(NULL);
693
694 scan_execution_context_t context;
695 init_scan_context(&context, options, start_time_milliseconds);
696
697 g_scan_interrupted = 0;
698 struct sigaction signal_action;
699 memset(&signal_action, 0, sizeof(signal_action));
700 signal_action.sa_handler = handle_interrupt_signal;
701 sigemptyset(&signal_action.sa_mask);
702
703 struct sigaction previous_sigint;
704 struct sigaction previous_sigterm;
705 sigaction(SIGINT, &signal_action, &previous_sigint);
706 sigaction(SIGTERM, &signal_action, &previous_sigterm);
707
708 bool execution_succeeded = config->target_count > 0;
709 scanner_target_t *targets = NULL;
710 size_t target_count = 0;
711 if (execution_succeeded) execution_succeeded = scanner_targets_prepare(config, &targets, &target_count);
712 bool database_transaction_open = false;
713 if (database && !options->dry_run) {
714 database_transaction_open = db_begin(database);
715 if (!database_transaction_open) {
716 context.database_failed = true;
717 execution_succeeded = false;
718 }
719 }
720 bool worker_failed = false;
721 int64_t pending_task_count = 1; /* Main-thread sentinel prevents shutdown while targets are queued. */
722 task_queue_t work_queue;
723 task_queue_t *active_work_queue = NULL;
724 pthread_t worker_threads[MAXIMUM_WORKER_COUNT];
725 scan_worker_context_t worker_contexts[MAXIMUM_WORKER_COUNT];
726 int started_worker_count = 0;
727 scan_worker_context_t common_worker_context = {
728 .work_queue = NULL,
729 .config = config,
730 .read_database = database,
731 .write_database = database,
732 .options = options,
733 .statistics = output_statistics,
734 .context = &context,
735 .worker_failed = &worker_failed,
736 .pending_task_count = &pending_task_count,
737 };
738
739 if (execution_succeeded && worker_count > DEFAULT_WORKER_COUNT) {
740 execution_succeeded = queue_initialize(&work_queue, DEFAULT_QUEUE_CAPACITY);
741 if (execution_succeeded) {
742 active_work_queue = &work_queue;
743 common_worker_context.work_queue = active_work_queue;
744 while (started_worker_count < worker_count) {
745 worker_contexts[started_worker_count] = common_worker_context;
746 if (database) {
747 worker_contexts[started_worker_count].read_database = db_open_read_only(database);
748 if (!worker_contexts[started_worker_count].read_database) break;
749 }
750 int thread_result = pthread_create(&worker_threads[started_worker_count], NULL, scan_worker,
751 &worker_contexts[started_worker_count]);
752 if (thread_result != 0) {
753 fprintf(stderr, "Failed to start scan worker: %s\n", strerror(thread_result));
754 if (database) db_close(worker_contexts[started_worker_count].read_database);
755 break;
756 }
757 started_worker_count++;
758 }
759 execution_succeeded = started_worker_count == worker_count;
760 }
761 }
762
763 if (config->target_count == 0) {
764 fprintf(stderr, "No targets configured. Pass target directories as arguments.\n");
765 } else if (execution_succeeded) {
766 for (size_t index = 0; index < target_count; index++) {
767 if (g_scan_interrupted) break;
768 const char *target = targets[index].path;
769 if (options->verbose) printf("Scanning target: %s\n", target);
770 bool target_succeeded;
771 if (active_work_queue) {
772 directory_task_t *directory_task = create_directory_task(target);
773 target_succeeded = directory_task != NULL;
774 if (target_succeeded) {
775 (void)__atomic_fetch_add(&pending_task_count, 1, __ATOMIC_ACQ_REL);
776 target_succeeded = queue_push(active_work_queue, directory_task);
777 if (!target_succeeded) {
778 free(directory_task);
779 complete_pending_task(&common_worker_context);
780 }
781 }
782 } else {
783 target_succeeded = process_path_entries(&common_worker_context, target);
784 }
785 if (!target_succeeded) {
786 execution_succeeded = false;
787 break;
788 }
789 }
790 }
791
792 if (active_work_queue) {
793 complete_pending_task(&common_worker_context);
794 if (!execution_succeeded || g_scan_interrupted) queue_signal_shutdown(active_work_queue);
795 for (int index = 0; index < started_worker_count; index++) {
796 pthread_join(worker_threads[index], NULL);
797 if (database) db_close(worker_contexts[index].read_database);
798 }
799 queue_destroy(active_work_queue);
800 if (__atomic_load_n(&worker_failed, __ATOMIC_RELAXED)) execution_succeeded = false;
801 }
802 scanner_targets_free(targets, target_count);
803 if (context.is_interactive_terminal && !options->verbose) {
804 printf("\r\033[K");
805 fflush(stdout);
806 }
807
808 if (g_scan_interrupted && !context.silent)
809 printf("\nScan interrupted by user. Saved partial progress to database.\n");
810
811 if (database_transaction_open) {
812 if (!db_commit(database)) context.database_failed = true;
813 context.pending_database_records = 0;
814 }
815
816 sigaction(SIGINT, &previous_sigint, NULL);
817 sigaction(SIGTERM, &previous_sigterm, NULL);
818
819 int64_t end_time_milliseconds = current_time_milliseconds();
820 int64_t end_time_unix = (int64_t)time(NULL);
821 output_statistics->duration_milliseconds = end_time_milliseconds - start_time_milliseconds;
822
823 if (database && !options->dry_run && output_statistics->files_scanned > 0) {
824 run_record_t run;
825 memset(&run, 0, sizeof(run));
826 run.started_at = start_time_unix;
827 run.completed_at = end_time_unix;
828 run.files_scanned = output_statistics->files_scanned;
829 run.files_compressed = output_statistics->files_compressed;
830 run.bytes_saved = output_statistics->bytes_saved;
831 run.duration_milliseconds = output_statistics->duration_milliseconds;
832 if (!db_record_run(database, &run)) context.database_failed = true;
833 }
834
835 return execution_succeeded && !g_scan_interrupted && !__atomic_load_n(&context.database_failed, __ATOMIC_RELAXED);
836}
