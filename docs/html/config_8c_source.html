70char *config_expand_path(const char *path) {
71 if (!path) return NULL;
72 if (path[0] == '~' && (path[1] == '/' || path[1] == '\0')) {
73 const char *home_directory = getenv("HOME");
74 if (!home_directory) {
75 struct passwd *password_entry = getpwuid(getuid());
76 if (password_entry) home_directory = password_entry->pw_dir;
77 }
78 if (home_directory) {
79 size_t home_length = strlen(home_directory);
80 size_t path_length = strlen(path + 1);
81 char *expanded_path = malloc(home_length + path_length + 1);
82 if (!expanded_path) return NULL;
83 strcpy(expanded_path, home_directory);
84 strcat(expanded_path, path + 1);
85 return expanded_path;
86 }
87 }
88 return strdup(path);
89}

148 if (!resolved_path) return config_create_default();

155

158 fseek(file_handle, 0, SEEK_SET);

161 fclose(file_handle);
162 free(resolved_path);
163 return config_create_default();
164 }

322bool config_remove_target(config_t *config, const char *path) {
323 if (!config || !path || config->target_count == 0) return false;
324
325 char *expanded_path = config_expand_path(path);
326 if (!expanded_path) return false;
327
328 char resolved_path[PATH_MAX];
329 const char *lookup_path = expanded_path;
330 if (realpath(expanded_path, resolved_path) != NULL) lookup_path = resolved_path;
331
332 size_t match_index = (size_t)-1;
333 for (size_t i = 0; i < config->target_count; i++) {
334 if (strcmp(config->targets[i], lookup_path) == 0 || strcmp(config->targets[i], expanded_path) == 0
335 || strcmp(config->targets[i], path) == 0) {
336 match_index = i;
337 break;
338 }
339 }
340 free(expanded_path);
341
342 if (match_index == (size_t)-1) return false;
343
344 free(config->targets[match_index]);
345 for (size_t i = match_index; i + 1 < config->target_count; i++)
346 config->targets[i] = config->targets[i + 1];
347 config->target_count--;
348
349 if (config->target_count == 0) {
350 free(config->targets);
351 config->targets = NULL;
352 }
353 return true;
